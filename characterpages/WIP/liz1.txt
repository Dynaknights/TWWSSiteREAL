// Simple pentatonic/city-pop inspired repeating sequence
function playCitypopLoop() {
  if (!audioCtx) return;
  
  // Notes in Hz for a citypop-ish 8-bit melody (Amaj pentatonic vibes)
  const melody = [
    {f:440, d:0.25}, {f:494, d:0.25}, {f:554, d:0.5},
    {f:587, d:0.25}, {f:659, d:0.25}, {f:587, d:0.25}, {f:554, d:0.25},
    {f:494, d:0.5},  {f:440, d:0.5},
    {f:370, d:0.25}, {f:415, d:0.25}, {f:440, d:0.5},
    {f:494, d:0.25}, {f:554, d:0.25}, {f:494, d:0.25}, {f:440, d:0.25},
    {f:370, d:1.0},
  ];
  
  // Chord bass notes
  const bass = [
    {f:110, d:1.0}, {f:110, d:1.0}, {f:98,  d:1.0}, {f:98, d:1.0},
    {f:110, d:1.0}, {f:110, d:1.0}, {f:98,  d:1.0}, {f:82, d:1.0},
  ];

  function playNote(freq, startTime, duration, type='square', gainVal=0.12, pan=0) {
    const osc  = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    const panner = audioCtx.createStereoPanner();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, startTime);
    gain.gain.setValueAtTime(0, startTime);
    gain.gain.linearRampToValueAtTime(gainVal, startTime + 0.01);
    gain.gain.setValueAtTime(gainVal, startTime + duration - 0.05);
    gain.gain.linearRampToValueAtTime(0, startTime + duration);
    panner.pan.value = pan;
    osc.connect(gain);
    gain.connect(panner);
    panner.connect(masterGain);
    osc.start(startTime);
    osc.stop(startTime + duration);
    musicNodes.push(osc);
  }

  function scheduleLoop() {
    const now = audioCtx.currentTime + 0.1;
    let t = now;
    melody.forEach(n => {
      playNote(n.f * 2, t, n.d * 0.5, 'square', 0.08, -0.3);
      t += n.d * 0.5;
    });
    const loopLen = melody.reduce((a,n) => a + n.d * 0.5, 0);
    
    let tb = now;
    bass.forEach(n => {
      playNote(n.f, tb, n.d * (loopLen / bass.length) * 0.9, 'sawtooth', 0.05, 0.3);
      tb += n.d * (loopLen / bass.length);
    });

      const hpf = audioCtx.createBiquadFilter();
      hpf.type = 'highpass';
      hpf.frequency.value = 8000;
      src.buffer = buffer;
      hiGain.gain.setValueAtTime(0.03, startT);
      hiGain.gain.linearRampToValueAtTime(0, startT + dur);
      src.connect(hpf);
      hpf.connect(hiGain);
      hiGain.connect(masterGain);
      src.start(startT);
      src.stop(startT + dur);
    }
    for (let i = 0; i < loopLen / 0.25; i++) {
      if (i % 2 === 0) makeNoise(now + i * 0.25, 0.08);
    }

    setTimeout(scheduleLoop, (loopLen - 0.5) * 1000);
  }

  scheduleLoop();
}

/* ============================================================
   THREE.JS — SOFT BOKEH DRIFT BACKGROUND
   ============================================================ */
(function() {
  const canvas = document.createElement('canvas');
  canvas.style.cssText = 'position:fixed;inset:0;z-index:1;pointer-events:none;opacity:0.55;';
  document.body.appendChild(canvas);

  const renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true });
  renderer.setClearColor(0x000000, 0);

  const scene = new THREE.Scene();
  const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0.1, 100);
  camera.position.z = 5;

  function resize() {
    renderer.setSize(window.innerWidth, window.innerHeight);
  }
  resize();
  window.addEventListener('resize', resize);

  // Create soft bokeh orbs
  const orbCount = 36;
  const orbs = [];
  const orbColors = [
    new THREE.Color('#FFB6C1'),
    new THREE.Color('#B4A0FF'),
    new THREE.Color('#FFE082'),
    new THREE.Color('#A8E6CF'),
    new THREE.Color('#FF9BB5'),
    new THREE.Color('#D4B0FF'),
  ];

  for (let i = 0; i < orbCount; i++) {
    const geo = new THREE.CircleGeometry(0.04 + Math.random() * 0.09, 32);
    const mat = new THREE.MeshBasicMaterial({
      color: orbColors[i % orbColors.length],
      transparent: true,
      opacity: 0.08 + Math.random() * 0.14,
    });
    const mesh = new THREE.Mesh(geo, mat);
    mesh.position.set((Math.random() - 0.5) * 2, (Math.random() - 0.5) * 2, 0);
    const speed = 0.0002 + Math.random() * 0.0004;
    const phase = Math.random() * Math.PI * 2;
    const ampX  = 0.15 + Math.random() * 0.25;
    const ampY  = 0.1  + Math.random() * 0.2;
    orbs.push({ mesh, speed, phase, ampX, ampY, ox: mesh.position.x, oy: mesh.position.y });
    scene.add(mesh);
  }

  let t = 0;
  function animate() {
    requestAnimationFrame(animate);
    t++;
    orbs.forEach((orb, i) => {
      orb.mesh.position.x = orb.ox + Math.sin(t * orb.speed + orb.phase) * orb.ampX;
      orb.mesh.position.y = orb.oy + Math.cos(t * orb.speed * 0.7 + orb.phase) * orb.ampY;
    });
    renderer.render(scene, camera);
  }
  animate();
})();
